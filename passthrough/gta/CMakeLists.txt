cmake_minimum_required(VERSION 3.20)
project(OrbitalPassthrough LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(OCP_BUILD_TESTS "Build the unit tests (portable core, needs no GTA V)" ON)
set(SHV_SDK_DIR "" CACHE PATH "ScriptHookV SDK root (contains inc/main.h and lib/ScriptHookV.lib)")
option(OCP_USE_STUB_SDK "Build/link the .asi against the bundled SDK stub (CI compile check only; the result is inert)" OFF)

if(MSVC)
  add_compile_options(/W4 /permissive- /Zc:__cplusplus /utf-8)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wshadow)
endif()

# ---- portable core -------------------------------------------------------------------------------------------------
file(GLOB OCP_CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp)
add_library(ocp_core STATIC ${OCP_CORE_SOURCES})
target_include_directories(ocp_core PUBLIC src ${CMAKE_CURRENT_SOURCE_DIR}/../protocol)
find_package(Threads REQUIRED)
target_link_libraries(ocp_core PUBLIC Threads::Threads)

# ---- the ScriptHookV plugin (Windows only) ---------------------------------------------------------------------------
if(WIN32 AND (SHV_SDK_DIR OR OCP_USE_STUB_SDK))
  add_library(OrbitalPassthrough SHARED src/shv/natives.cpp src/shv/script.cpp src/shv/dllmain.cpp)
  if(OCP_USE_STUB_SDK)
    target_include_directories(OrbitalPassthrough PRIVATE tests/shv_stub)
    target_sources(OrbitalPassthrough PRIVATE tests/shv_stub/shv_stub_impl.cpp)
  else()
    target_include_directories(OrbitalPassthrough PRIVATE ${SHV_SDK_DIR}/inc)
    target_link_libraries(OrbitalPassthrough PRIVATE ${SHV_SDK_DIR}/lib/ScriptHookV.lib)
  endif()
  target_link_libraries(OrbitalPassthrough PRIVATE ocp_core)
  # ScriptHookV loads *.asi from the game directory.
  set_target_properties(OrbitalPassthrough PROPERTIES PREFIX "" SUFFIX ".asi")
  if(MSVC)
    target_compile_options(OrbitalPassthrough PRIVATE /O2 /GL)
    target_link_options(OrbitalPassthrough PRIVATE /LTCG)
  endif()
elseif(WIN32)
  message(STATUS "Set -DSHV_SDK_DIR=<ScriptHookV SDK> to build OrbitalPassthrough.asi")
endif()

# ---- tests -----------------------------------------------------------------------------------------------------------
if(OCP_BUILD_TESTS AND NOT WIN32)
  enable_testing()
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE ocp_core)
  add_executable(test_app tests/test_app.cpp)
  target_link_libraries(test_app PRIVATE ocp_core)
  add_executable(test_natives tests/test_natives.cpp src/shv/natives.cpp)
  target_include_directories(test_natives PRIVATE tests/shv_stub)
  target_link_libraries(test_natives PRIVATE ocp_core)
  add_executable(fake_gta tests/fake_gta.cpp)
  target_link_libraries(fake_gta PRIVATE ocp_core)
  add_executable(latency_bench tests/latency_bench.cpp)
  target_link_libraries(latency_bench PRIVATE ocp_core)
  add_executable(layout_dump tests/layout_dump.cpp)
  target_include_directories(layout_dump PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../protocol)
  add_test(NAME core COMMAND test_core)
  add_test(NAME app COMMAND test_app)
  add_test(NAME natives COMMAND test_natives)
endif()
